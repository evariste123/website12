    // Form submit - improved validation
    volForm.addEventListener('submit', function(evt){ 
      evt.preventDefault(); 
      formmsg.textContent=''; 
      formmsg.style.color=''; 
      
      const name = document.getElementById('name').value.trim(); 
      const email = document.getElementById('email').value.trim(); 
      const phone = document.getElementById('phone').value.trim(); 
      const role = document.getElementById('role').value.trim(); 
      const availability = document.getElementById('availability').value.trim(); 
      const skills = document.getElementById('skills').value.trim(); 
      const bio = document.getElementById('bio').value.trim(); 
      const agree = document.getElementById('agree').checked; 
      const lang = langSelect.value; 
      
      console.log('Form submit:', {name,email,role,availability,bio,agree});
      
      // Validation
      if(!name||!email||!role||!availability||!bio){ 
        formmsg.textContent = 'Please fill all required fields'; 
        formmsg.style.color='#ffd9d9'; 
        return; 
      } 
      if(bio.length < 50){ 
        formmsg.textContent = 'Bio must be at least 50 characters'; 
        formmsg.style.color='#ffd9d9'; 
        return; 
      } 
      const emailRe = /^[^\s@]+@[^\s@]+\.[^\s@]+$/; 
      if(!emailRe.test(email)){ 
        formmsg.textContent = 'Please enter a valid email'; 
        formmsg.style.color='#ffd9d9'; 
        return; 
      } 
      if(!agree){ 
        formmsg.textContent = 'You must agree to be contacted'; 
        formmsg.style.color='#ffd9d9'; 
        return; 
      } 
      
      // Save application
      const app = {name,email,phone,role,availability,skills,bio,submitted:new Date().toISOString(), lang}; 
      try{ 
        const existing = JSON.parse(localStorage.getItem(APPS)||'[]'); 
        existing.push(app); 
        localStorage.setItem(APPS, JSON.stringify(existing)); 
        console.log('Application saved'); 
        setAccess(true); 
        unlockAnimation(t(lang,'accessGranted')||'Access granted!'); 
        volForm.reset();
        hideOverlayAfter(900); 
      }catch(err){ 
        console.error('Save error:', err); 
        formmsg.textContent='Error saving form'; 
        formmsg.style.color='#ffd9d9'; 
      } 
    });
